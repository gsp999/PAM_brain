import unittest
from unittest.mock import patch
import tempfile
from pathlib import Path
from types import SimpleNamespace
import json
import numpy as np
from pam_mosaic.backend.poses import Edge, _lm_residual_jacobian, pose_to_matrix
from pam_mosaic.backend.window import schur_prior, optimize_window
from pam_mosaic.backend.keyframes import BackendConfig, run_keyframe_backend
from pam_mosaic.backend.factors import solve_measurement, transform
from pam_mosaic.frontend.registration import Match
from pam_mosaic.pipeline import run
from pam_mosaic.preprocessing.dataset import Record
from pam_mosaic.preprocessing.inputs import InputState


class KeyframeBackendTests(unittest.TestCase):
    def test_pipeline_optional_backend_writes_auditable_result(self):
        frame=np.full((32,32),.1,np.float32)
        adapter=SimpleNamespace(DISPLAY_RANGE=(0.,.25),
            prepare=lambda paths:InputState(),options=lambda enabled,excess:{},
            load_frame=lambda path,state,options:(frame.copy(),{}))
        with tempfile.TemporaryDirectory() as directory:
            root=Path(directory);records=[]
            for i in range(10):
                path=root/f'{i}.dat';path.touch();records.append(Record(i+1,1,path))
            with patch('pam_mosaic.pipeline.get_format',return_value=adapter), \
                 patch('pam_mosaic.pipeline.candidate_pairs',return_value=[(i,i+1) for i in range(9)]), \
                 patch('pam_mosaic.pipeline.register',return_value=Match(np.eye(2,3),30,0.,1.,True)):
                report=run(records,root/'result',backend_factors=('geometry','intensity'),background_correction=False)
            self.assertEqual(len(report['components']),1)
            self.assertEqual(report['components'][0]['frames'],10)
            metadata=json.loads((root/'result/component_001/backend.json').read_text())
            self.assertEqual(metadata['keyframes'],[0,8,9])
            self.assertEqual(len(metadata['ordinary_frame_references']),10)
            self.assertTrue((root/'result/backend_group_001.json').exists())

    def test_information_weighted_analytic_jacobian(self):
        poses=np.array([[0.,0.,0.],[4.,2.,.01],[9.,3.,.03]])
        edge=Edge(1,2,np.array([[1.,0.,5.],[0.,1.,1.]]),.4,'geometry',np.array([[1.,.1,0],[0,.5,.2],[0,0,.3]]))
        r,j=_lm_residual_jacobian(poses,[edge]);numeric=[]
        for i in range(6):
            a=poses.copy();b=poses.copy()
            a[1+i//3,i%3]+=1e-6;b[1+i//3,i%3]-=1e-6
            numeric.append((_lm_residual_jacobian(a,[edge])[0]-_lm_residual_jacobian(b,[edge])[0])/2e-6)
        np.testing.assert_allclose(j.toarray(),np.column_stack(numeric),atol=1e-7)

    def test_schur_matches_joint_linear_solution(self):
        rng=np.random.default_rng(4)
        a=rng.normal(size=(20,9));h=a.T@a+np.eye(9);g=rng.normal(size=9)
        sh,sg=schur_prior(h,g,6)
        np.testing.assert_allclose(np.linalg.solve(sh,-sg),np.linalg.solve(h,-g)[6:],atol=1e-10)

    def test_window_keeps_old_states_and_anchor(self):
        poses=np.array([[i*3.,0.,0.] for i in range(8)])
        edges=[Edge(i,i+1,np.array([[1.,0.,3.],[0.,1.,0.]])) for i in range(7)]
        poses[-2:,0]+=1
        result=optimize_window(poses,edges,2)
        np.testing.assert_array_equal(result[:-2],poses[:-2])
        np.testing.assert_allclose(result[-2:,0],[18.,21.],atol=1e-4)

    def test_measurement_recovers_points_and_rejects_unobservable_line(self):
        rng=np.random.default_rng(3);src=rng.uniform(0,200,(60,2))
        truth=pose_to_matrix(np.array([2.,-1.,.005]))
        dst=transform(src,truth)
        result,audit=solve_measurement(np.eye(3),lambda m:(transform(src,m)-dst).ravel(),'intensity',48)
        self.assertTrue(audit['accepted'])
        np.testing.assert_allclose(result[0],truth,atol=1e-5)
        result,audit=solve_measurement(np.eye(3),lambda m:transform(src,m)[:,0]-dst[:,0],'geometry',40)
        self.assertIsNone(result)
        self.assertFalse(audit['accepted'])

    def test_ordinary_frames_follow_keys_and_disconnections_split(self):
        rng=np.random.default_rng(2)
        frames=[rng.uniform(.02,.25,(32,32)).astype(np.float32) for _ in range(9)]
        edges=[Edge(i,i+1,np.array([[1.,0.,3.],[0.,1.,0.]])) for i in range(8) if i!=5]
        invalid=Match(np.eye(2,3),0,float('inf'),0.,False)
        with patch('pam_mosaic.backend.keyframes.register',return_value=invalid):
            runs,audit=run_keyframe_backend(frames,edges,BackendConfig((),3,2,1.,10.,30))
        self.assertEqual([len(r['frame_indices']) for r in runs],[6,3])
        for r in runs:
            np.testing.assert_allclose(r['poses'][:,0],np.arange(len(r['poses']))*3,atol=1e-7)
            np.testing.assert_allclose(r['poses'][:,1:],0,atol=1e-7)
        self.assertEqual(runs[0]['ordinary_frame_references'][1],0)
        self.assertTrue(any(a['reason']=='new_submap_no_past_match' for a in audit['measurements']))


if __name__=='__main__':
    unittest.main()
